cmake_minimum_required(VERSION 3.24)
project(QueenContentContract LANGUAGES C CXX)
set(queen_root "${CMAKE_CURRENT_LIST_DIR}/../../..")
set(ARP_THIRD_PARTY_DIR "${queen_root}/3rd")
include("${queen_root}/cmake/ArpMimalloc.cmake")
add_library(arp_core STATIC "${queen_root}/src/core/memory/cpu_allocator.cpp"
    "${queen_root}/src/core/encoding/base64.cpp")
target_include_directories(arp_core PUBLIC "${queen_root}/src")
target_compile_definitions(arp_core PUBLIC ARP_STATIC_LINK=1)
target_compile_features(arp_core PUBLIC cxx_std_17)
target_link_libraries(arp_core PUBLIC arp_mimalloc)
include("${queen_root}/cmake/ArpQueenContent.cmake")
arp_add_queen_content()
add_executable(queen_content_tests content_tests.cpp)
target_link_libraries(queen_content_tests PRIVATE arp_queen_content)
target_compile_options(queen_content_tests PRIVATE -UNDEBUG)
if(MINGW)
    target_link_options(queen_content_tests PRIVATE -static-libgcc -static-libstdc++)
endif()
enable_testing()
add_test(NAME queen_content_contract COMMAND queen_content_tests "${queen_root}/examples/build_a_queen/assets/characters")
add_test(NAME queen_content_game_smoke COMMAND queen_content_tests "${queen_root}/examples/build_a_queen/assets/characters" --smoke)
